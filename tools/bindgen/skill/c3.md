A test function (`c3c test`), calling the driver through `raven::c`. Texts from Raven are
`ZString`s:

```c3
import raven::c;

fn ulong find(ZString selector)
{
	c::r_wait_idle(600);
	ulong found = c::r_accessible_find(0, selector);
	assert(found, "%s: %s", selector, (ZString)c::r_accessible_get_last_error());
	return found;
}

fn void acted(CInt done)
{
	assert(done, "%s", (ZString)c::r_accessible_get_last_error());
	c::r_wait_idle(600);
}

fn void adds_a_note() @test
{
	RInitDesc desc = { .backend = c::R_BACKEND_NULL, .headless = c::R_TRUE };
	assert(c::r_init_ex(&desc), "Raven didn't start");
	defer c::r_shutdown();
	c::r_set_animation_time_step(1.0 / 60);
	open_notes("notes.xml"); // the app's code: builds the window, binds its actions
	acted(c::r_accessible_type(find("role=text-field"), "buy milk"));
	acted(c::r_accessible_click(find("role=button name=\"Add\"")));
	assert(c::r_accessible_count(0, "role=list-item text=\"buy milk\"") == 1, "no note \"buy milk\"");
}
```
